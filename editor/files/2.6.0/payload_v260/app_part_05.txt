    def _apply_file_manifest_update(self, m):
        remote = str(m.get("version", "")).strip()
        files = m.get("files") or []
        if not remote or not files:
            raise RuntimeError("Dosya tabanlı güncelleme bilgisi eksik.")

        stage = Path(tempfile.gettempdir()) / f"SolinajEditor_files_{remote}_{int(time.time())}"
        if stage.exists():
            shutil.rmtree(stage, ignore_errors=True)
        stage.mkdir(parents=True, exist_ok=True)

        total = len(files)
        for i, item in enumerate(files, 1):
            rel = str(item.get("path", "")).replace("\\", "/").lstrip("/")
            file_url = str(item.get("url", "")).strip()
            if not rel or not file_url or ".." in Path(rel).parts:
                raise RuntimeError(f"Geçersiz güncelleme dosyası: {rel}")

            target = stage / Path(rel)
            target.parent.mkdir(parents=True, exist_ok=True)

            sep = "&" if "?" in file_url else "?"
            fresh_url = f"{file_url}{sep}v={remote}&ts={int(time.time())}"
            req = urllib.request.Request(
                fresh_url,
                headers={
                    "User-Agent": "SolinajEditorUpdater",
                    "Cache-Control": "no-cache",
                    "Pragma": "no-cache"
                }
            )
            with urllib.request.urlopen(req, timeout=120) as r, open(target, "wb") as f:
                shutil.copyfileobj(r, f)

            expected = str(item.get("sha256", "")).strip().lower()
            if expected:
                got = hashlib.sha256(target.read_bytes()).hexdigest()
                if got != expected:
                    raise RuntimeError(f"{rel} doğrulanamadı.")

            pct = 10 + (i / max(1, total)) * 75
            self.after(0, lambda p=pct, n=i, t=total, rv=remote:
                self._set_update_ui(
                    f"V{rv} dosyaları indiriliyor...",
                    f"Dosya {n}/{t}",
                    p, title="Güncelleme indiriliyor", badge=""))

        root = Path(__file__).resolve().parent
        self.after(0, lambda r=remote:
            self._set_update_ui(
                f"V{r} indirildi. Kurulum başlatılıyor...",
                "Kuruluyor • program birazdan kapanacak",
                92, title="Güncelleme kuruluyor", badge=""))

        subprocess.Popen([
            "powershell.exe","-NoProfile","-ExecutionPolicy","Bypass","-File",
            str(root/"AUTO_UPDATE_FILES_APPLY.ps1"),
            "-StageDir",str(stage),
            "-InstallDir",str(root),
            "-Launcher",str(root/"SolinajEditor_Launcher.vbs"),
            "-PythonExe",sys.executable,
            "-TargetVersion",remote
        ], creationflags=getattr(subprocess,"CREATE_NO_WINDOW",0))

        self.after(0, lambda:
            self._set_update_ui(
                "Dosyalar kuruluyor. SOLINAJ EDITOR otomatik yeniden açılacak.",
                "Kurulum başladı ✓",
                100, title="Kurulum başladı", badge="✓ Yeniden başlatılacak"))
        self.after(900, self.destroy)

    def _update_worker(self):
        self.after(0, lambda: self.status.set("Güncelleme kontrol ediliyor..."))
        try:
            cfg=json.loads((Path(__file__).resolve().parent/"update_config.json").read_text(encoding="utf-8-sig"))
            url=str(cfg.get("manifest_url","")).strip()
            if not url or url.startswith("BURAYA_"):
                self.after(0, lambda: self._set_update_ui("Güncelleme adresi tanımlı değil.", "İşlem durduruldu", 0))
                self.after(0, lambda: messagebox.showinfo("Güncelleme",
                    "İnternetten otomatik güncelleme için kalıcı güncelleme adresi tanımlanması gerekiyor."))
                self.after(0, lambda: self.status.set("Hazır"))
                self.after(0, self._finish_update_check)
                return

            req=urllib.request.Request(url,headers={"User-Agent":"SolinajEditorUpdater","Cache-Control":"no-cache"})
            with urllib.request.urlopen(req,timeout=20) as r:
                m=json.loads(r.read().decode("utf-8-sig"))

            remote=str(m.get("version","0.0.0"))
            def vt(v):
                return tuple(int(x) if x.isdigit() else 0 for x in (v.split(".")+["0","0"])[:3])

            if vt(remote)<=vt(APP_VERSION):
                self.after(0, lambda: self._set_update_ui(
                    f"Kurulu sürüm V{APP_VERSION}. Yeni bir güncelleme bulunamadı.", "Güncel ✓", 100, title="Program güncel", badge=f"✓ Program Güncel — V{APP_VERSION}"))
                self.after(0, lambda: messagebox.showinfo("Güncelleme",
                    f"En güncel sürümü kullanıyorsun.\n\nKurulu sürüm: V{APP_VERSION}"))
                self.after(0, lambda: self.status.set("Hazır"))
                self.after(0, self._finish_update_check)
                return

            if m.get("files"):
                self.after(0, lambda r=remote: self.status.set(f"V{r} bulundu • dosya güncellemesi hazırlanıyor..."))
                return self._apply_file_manifest_update(m)

            package=str(m.get("package_url","")).strip()
            if not package:
                raise RuntimeError("Güncelleme paket adresi yok.")

            notes=str(m.get("notes","")).strip()
            info = f"Yeni sürüm bulundu: V{remote}"
            if notes:
                info += f"\n{notes[:180]}"
